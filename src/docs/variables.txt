========= LK ========= || ========= CPP =========

i8   u8                || int8_t   uint8_t
i16  u16               || int16_t  uint16_t
i32  u32  f32          || int32_t  uint32_t  float
i64  u64  f64          || int64_t  uint64_t  double

str                    || string
set  uset              || set     unordered_set
map  umap              || map     unordered_map

type a                 || ініціалізація дефолтним значенням
type a : b             || ініціалізація
type a : (b1, b2, ..)  || ініціалізація типу із множиною параметрів
#type a : b            || константа
type[] a : b1, b2, ..  || масив
str a : ".."           || рядок

Приклад: a - примітив, b - структура

foo(a)                 || копіювання
foo(.a)                || посилання
foo(:a)                || ERROR (немає сенсу)

foo(b)                 || посилання (покажчик під капотом)
foo(.b)                || ERROR (проблема двійного покажчика)
foo(:b)                || переміщення (переміщення власника)
